    function publishRegexScanBatch(generation, javaIds, scanned,
            total, complete, cacheSignature) {
        var ids;
        var index;
        if (generation !== regexScanGeneration || !ready) { return false; }
        ids = copyJavaIdList(javaIds);
        for (index = 0; index < ids.length; index += 1) {
            if (!regexMatchedIdSet[String(ids[index])]) {
                regexMatchedIdSet[String(ids[index])] = true;
                regexMatchedIds.push(ids[index]);
            }
        }
        regexScanState.scanned = Number(scanned);
        regexScanState.total = Number(total);
        regexScanState.matched = regexMatchedIds.length;
        if (ids.length > 0 && !regexScanState.firstBatchPublished) {
            regexScanState.firstBatchPublished = true;
        }
        if (regexScanState.firstBatchPublished || complete === true) {
            regexResultPageResult({ append: false,
                page: paginationState.mode === "number" ?
                    paginationState.pageNumber : 1 });
            refreshRegexResultUi(complete ?
                "regex_scan_complete" : "regex_scan_batch");
        } else {
            updateResultCountOnMain();
        }
        if (complete === true) {
            regexScanState.running = false;
            regexScanState.complete = true;
            regexScanFuture = null;
            paginationState.totalCount = regexMatchedIds.length;
            paginationState.totalPages = regexMatchedIds.length > 0 ?
                Math.ceil(regexMatchedIds.length /
                    Math.max(1, Number(paginationState.pageSize))) : 0;
            paginationState.hasMore = previewRows.length < regexMatchedIds.length;
            if (paginationState.repositoryMode === "number") {
                numberPagerState.loading = false;
                numberPagerState.totalCount = paginationState.totalCount;
                numberPagerState.totalPages = paginationState.totalPages;
            }
            regexResultCache[String(cacheSignature)] = {
                matchedIds: regexMatchedIds.slice(0),
                completedAt: ClipHub.Base.now()
            };
            refreshRegexResultUi("regex_scan_complete");
        }
        return true;
    }
