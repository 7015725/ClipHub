07136     function headerMetrics() {
07137         var widthDp = Number(state.panelWidthDp || 0);
07138         var fontScale = resourceFontScale();
07139         var touchDp = Math.max(1, Number(touchSlop || 1) / density);
07140         var baseDp;
07141         var actionSizeDp;
07142         var controlHeightDp;
07143         var gapDp;
07144         var titleSp;
07145         var iconSp;
07146         var statusSp;
07147         var searchSp;
07148         var radiusDp;
07149         var inputPaddingDp;
07150         var badgeSizeDp;
07151         var badgeSp;
07152         if (widthDp <= 0 && Number(state.panelWidthPx || 0) > 0) {
07153             widthDp = Number(state.panelWidthPx) / density;
07154         }
07155         if (widthDp <= 0) {
07156             widthDp = Number(appContext.getResources()
07157                 .getDisplayMetrics().widthPixels) / density;
07158         }
07159         baseDp = Math.max(touchDp, widthDp * 0.018);
07160         actionSizeDp = clampNumber(widthDp * 0.092,
07161             baseDp * 4.4, widthDp * 0.12);
07162         controlHeightDp = clampNumber(actionSizeDp * 1.02,
07163             baseDp * 4.6, widthDp * 0.125);
07164         gapDp = clampNumber(widthDp * 0.014,
07165             baseDp * 0.65, actionSizeDp * 0.24);
07166         titleSp = clampNumber(widthDp / (fontScale * 23),
07167             actionSizeDp / (fontScale * 2.45),
07168             actionSizeDp / (fontScale * 1.85));
07169         iconSp = clampNumber(actionSizeDp / (fontScale * 2.05),
07170             titleSp * 0.86, titleSp * 1.18);
07171         statusSp = clampNumber(titleSp * 0.60,
07172             iconSp * 0.58, titleSp * 0.72);
07173         searchSp = clampNumber(titleSp * 0.70,
07174             statusSp, titleSp * 0.82);
07175         radiusDp = Math.max(baseDp * 1.3, controlHeightDp * 0.44);
07176         inputPaddingDp = Math.max(baseDp * 0.65, gapDp);
07177         badgeSizeDp = Math.max(baseDp * 2.0, actionSizeDp * 0.38);
07178         badgeSp = Math.max(statusSp * 0.64,
07179             badgeSizeDp / (fontScale * 3.4));
07180         state.headerHeightDp = actionSizeDp + gapDp + controlHeightDp;
07181         state.headerControlHeightDp = controlHeightDp;
07182         state.headerActionSizeDp = actionSizeDp;
07183         state.headerGapDp = gapDp;
07184         return {
07185             widthDp: widthDp,
07186             fontScale: fontScale,
07187             baseDp: baseDp,
07188             actionSizeDp: actionSizeDp,
07189             controlHeightDp: controlHeightDp,
07190             gapDp: gapDp,
07191             titleSp: titleSp,
07192             iconSp: iconSp,
07193             statusSp: statusSp,
07194             searchSp: searchSp,
07195             radiusDp: radiusDp,
07196             inputPaddingDp: inputPaddingDp,
07197             badgeSizeDp: badgeSizeDp,
07198             badgeSp: badgeSp
07199         };
07200     }
